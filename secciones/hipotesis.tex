\section{Formulación de hipótesis}

\subsection*{Hipótesis 1: Tiempo real vs. tiempo estimado}
\begin{itemize}
    \item \textbf{$H_{1a}$}: El tiempo real de recorrido en el corredor supera, en promedio, al tiempo estimado por Google Maps.
    \item \textbf{$H_{0a}$}: No existe diferencia entre el tiempo real y el estimado por Google Maps en el corredor.
\end{itemize}

\subsection*{Hipótesis 2: Variación según la hora (pico vs. no pico)}
\begin{itemize}
    \item \textbf{$H_{1b}$}: La diferencia entre el tiempo real y el estimado es mayor en hora pico que en hora no pico.
    \item \textbf{$H_{0b}$}: La diferencia entre el tiempo real y el estimado es igual en hora pico y en hora no pico.
\end{itemize}

\subsection*{Hipótesis 3: Influencia del clima (lluvia vs. sin lluvia)}
\begin{itemize}
    \item \textbf{$H_{1c}$}: La diferencia entre el tiempo real y el estimado es mayor en recorridos con lluvia que en recorridos sin lluvia.
    \item \textbf{$H_{0c}$}: La diferencia entre el tiempo real y el estimado no varía según haya o no lluvia.
\end{itemize}
